cmake_minimum_required(VERSION 3.22)
project(frostbox LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(frostbox_process STATIC ../process/process.cpp ../wine/wine_runtime.cpp ../wine/android_wine_command.cpp ../cpu/box64_backend.cpp)
set_target_properties(frostbox_process PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(frostbox_process PUBLIC ../process ../wine ../cpu)
target_link_libraries(frostbox_process PUBLIC Threads::Threads)
target_compile_options(frostbox_process PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    add_library(frostbox_surface SHARED ../display/surface_owner.cpp ../../graphics/wsi/image_bridge.cpp ../../graphics/vulkan/capabilities.cpp)
    target_link_libraries(frostbox_surface PRIVATE frostbox_process vulkan android dl)
    target_compile_options(frostbox_surface PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    add_library(frostbox_vulkan SHARED ../../graphics/vulkan/capabilities.cpp ../../graphics/vulkan/renderer.cpp ../../graphics/vulkan/map_probe.cpp ../../graphics/vulkan/jni.cpp)
    target_link_libraries(frostbox_vulkan PRIVATE frostbox_process vulkan android log)
    target_compile_options(frostbox_vulkan PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    add_library(frostbox SHARED bridge.cpp process_jni.cpp)
    target_link_libraries(frostbox PRIVATE frostbox_process vulkan log)
    target_compile_options(frostbox PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
else()
    enable_testing()
    add_executable(runtime_contract_test contract_test.cpp)
    target_compile_options(runtime_contract_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME runtime_contract COMMAND runtime_contract_test)
    add_executable(frostbox_probe ../process/probe.cpp)
    add_executable(process_test ../process/process_test.cpp)
    target_link_libraries(process_test PRIVATE frostbox_process)
    target_compile_options(process_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME process_lifecycle COMMAND process_test $<TARGET_FILE:frostbox_probe>)
    set_tests_properties(process_lifecycle PROPERTIES TIMEOUT 15)
    add_executable(frostbox_wine_smoke ../wine/wine_smoke.cpp)
    target_link_libraries(frostbox_wine_smoke PRIVATE frostbox_process)
endif()
